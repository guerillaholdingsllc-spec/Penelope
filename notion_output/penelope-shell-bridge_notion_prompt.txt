        ```notion
        if(prop("Total Tasks") == 0, "0%", format(round(prop("Completed Tasks") / prop("Total Tasks") * 100)) + "% " + slice("■■■■■■■■■■", 0, floor(prop("Completed Tasks") / prop("Total Tasks") * 10)) + slice("□□□□□□□□□□", 0, 10 - floor(prop("Completed Tasks") / prop("Total Tasks") * 10)))
        ```
